    function showToast(message, type = 'success') {

    function handlePhotoUpload(input, previewImgId, wrapperId, bufferKey) {
      const file = input.files[0];
      if (!file) return;
      if (file.size > 3 * 1024 * 1024) {
        showToast('Ukuran foto melebihi batas maksimal 3 MB!', 'error');
        input.value = "";
        return;
      }
      const reader = new FileReader();
      reader.onload = function (e) {
        photoBuffers[bufferKey] = e.target.result;
        document.getElementById(previewImgId).src = e.target.result;
        document.getElementById(wrapperId).classList.remove('hidden');
        if (bufferKey === 'adm') {
          document.getElementById('adm-foto-label').textContent = file.name.slice(0, 16) + '...';
        }
        showToast('Foto dokumentasi kerusakan berhasil dimuat!', 'success');
      };
      reader.readAsDataURL(file);
    }

    function searchTicket() {
      const query = document.getElementById('track-input').value.trim().toLowerCase();
      const container = document.getElementById('track-result-container');
      if (!query) {
        container.innerHTML = `<div class="bg-amber-50 border border-amber-200 text-amber-800 p-4 rounded-2xl text-xs font-bold">Silakan masukkan Nomor Tiket terlebih dahulu.</div>`;
        return;
      }

      const found = reports.find(r => r.id.toLowerCase() === query || (r.nomorHp && r.nomorHp.toLowerCase().includes(query)));
      if (!found) {
        container.innerHTML = `
          <div class="bg-white border border-red-200 rounded-2xl p-6 text-center space-y-2 shadow-sm">
            <div class="w-12 h-12 rounded-full bg-red-50 text-red-600 flex items-center justify-center mx-auto text-xl"><i class="fa-solid fa-file-circle-question"></i></div>
            <h4 class="text-sm font-bold text-slate-800">Tiket Laporan Tidak Ditemukan</h4>
            <p class="text-xs text-slate-500">Pastikan Nomor Tiket benar (contoh: <span class="font-mono font-bold">LP-2026-00124</span>).</p>
          </div>`;
        return;
      }

    function getAggregatedStats() {
      const total = HISTORICAL_OFFSET.total + reports.length;
      const berat = HISTORICAL_OFFSET.berat + reports.filter(r => r.tingkatKerusakan === 'Rusak Berat').length;
      const sedang = HISTORICAL_OFFSET.sedang + reports.filter(r => r.tingkatKerusakan === 'Rusak Sedang').length;
      const ringan = HISTORICAL_OFFSET.ringan + reports.filter(r => r.tingkatKerusakan === 'Rusak Ringan').length;

      const verif = HISTORICAL_OFFSET.verif + reports.filter(r => r.status === 'Menunggu Verifikasi').length;
      const progress = HISTORICAL_OFFSET.progress + reports.filter(r => r.status === 'Dalam Penanganan').length;
      const done = HISTORICAL_OFFSET.done + reports.filter(r => r.status === 'Selesai').length;

      return { total, berat, sedang, ringan, verif, progress, done };
    }

    function renderDonutChart(s) {
      document.getElementById('donut-center-total').textContent = s.total;
      const labels = ['Rusak Berat', 'Rusak Sedang', 'Rusak Ringan', 'Menunggu Verifikasi', 'Dalam Penanganan', 'Selesai'];
      const values = [s.berat, s.sedang, s.ringan, s.verif, s.progress, s.done];
      const colors = ['#DC2626', '#EAB308', '#0284C7', '#7C3AED', '#0D9488', '#059669'];

      document.getElementById('donut-legend').innerHTML = labels.map((lbl, idx) => {
        const pct = ((values[idx] / s.total) * 100).toFixed(1);
        return `
          <div class="flex items-center justify-between">
            <div class="flex items-center gap-2">
              <span class="w-2.5 h-2.5 rounded-sm shrink-0" style="background:${colors[idx]}"></span>
              <span class="text-slate-600 truncate">${lbl}</span>
            </div>
            <span class="font-bold text-slate-800">${values[idx]} <span class="text-slate-400 font-normal">(${pct}%)</span></span>
          </div>
        `;
      }).join('');

      const ctx = document.getElementById('statusDonutChart').getContext('2d');
      if (statusChartInstance) {
        statusChartInstance.data.datasets[0].data = values;
        statusChartInstance.update();
      } else {
        statusChartInstance = new Chart(ctx, {
          type: 'doughnut',
          data: {
            labels: labels,
            datasets: [{
              data: values,
              backgroundColor: colors,
              borderWidth: 2,
              borderColor: '#ffffff'
            }]
          },
          options: {
            responsive: true,
            maintainAspectRatio: true,
            cutout: '66%',
            plugins: { legend: { display: false } }
          }
        });
      }
    }

    function getStatusBadge(status) {
      if (status === 'Menunggu Verifikasi') {
        return `<span class="px-2 py-0.5 rounded-md text-[10px] font-bold bg-purple-100 text-purple-800 border border-purple-200 whitespace-nowrap">Verifikasi</span>`;
      }
      if (status === 'Dalam Penanganan') {
        return `<span class="px-2 py-0.5 rounded-md text-[10px] font-bold bg-sky-100 text-sky-800 border border-sky-200 whitespace-nowrap">Penanganan</span>`;
      }
      return `<span class="px-2 py-0.5 rounded-md text-[10px] font-bold bg-emerald-100 text-emerald-800 border border-emerald-200 whitespace-nowrap">Selesai</span>`;
    }

    function getSeverityBadge(tingkat) {
      if (tingkat === 'Rusak Berat') {
        return `<span class="px-2 py-0.5 rounded-md text-[10px] font-bold bg-red-600 text-white whitespace-nowrap">Rusak Berat</span>`;
      }
      if (tingkat === 'Rusak Sedang') {
        return `<span class="px-2 py-0.5 rounded-md text-[10px] font-bold bg-amber-500 text-white whitespace-nowrap">Rusak Sedang</span>`;
      }
      return `<span class="px-2 py-0.5 rounded-md text-[10px] font-bold bg-sky-600 text-white whitespace-nowrap">Rusak Ringan</span>`;
    }

    function getDistrikSummary() {
      const base = [
        { distrik: "Tigi", total: 31, berat: 9, sedang: 14, ringan: 8 },
        { distrik: "Tigi Timur", total: 27, berat: 7, sedang: 10, ringan: 10 },
        { distrik: "Tigi Barat", total: 23, berat: 6, sedang: 8, ringan: 9 },
        { distrik: "Bowobado", total: 19, berat: 4, sedang: 8, ringan: 7 },
        { distrik: "Kapiraya", total: 16, berat: 3, sedang: 5, ringan: 8 }
      ];

      reports.forEach(r => {
        let row = base.find(b => b.distrik.toLowerCase() === r.distrik.toLowerCase());
        if (!row) {
          row = { distrik: r.distrik, total: 0, berat: 0, sedang: 0, ringan: 0 };
          base.push(row);
        }
        row.total += 1;
        if (r.tingkatKerusakan === 'Rusak Berat') row.berat += 1;
        else if (r.tingkatKerusakan === 'Rusak Sedang') row.sedang += 1;
        else row.ringan += 1;
      });

      return base.sort((a, b) => b.total - a.total);
    }

    function renderFullDataTable() {
      const q = document.getElementById('data-search-input').value.trim().toLowerCase();
      const distrik = document.getElementById('data-filter-distrik').value;
      const kampung = document.getElementById('data-filter-kampung').value;
      const severity = document.getElementById('data-filter-severity').value;
      const status = document.getElementById('data-filter-status').value;

      const filtered = reports.filter(r => {
        const matchQ = !q || r.id.toLowerCase().includes(q) || r.namaPelapor.toLowerCase().includes(q) || r.namaRuas.toLowerCase().includes(q) || r.kampung.toLowerCase().includes(q);
        const matchD = distrik === 'ALL' || r.distrik === distrik;
        const matchK = kampung === 'ALL' || r.kampung === kampung;
        const matchSev = severity === 'ALL' || r.tingkatKerusakan === severity;
        const matchSt = status === 'ALL' || r.status === status;
        return matchQ && matchD && matchK && matchSev && matchSt;
      });

      const tbody = document.getElementById('full-data-tbody');
      if (filtered.length === 0) {
        tbody.innerHTML = `<tr><td colspan="9" class="py-8 text-center text-slate-500">Tidak ada data laporan yang sesuai dengan filter pencarian.</td></tr>`;
        return;
      }

      tbody.innerHTML = filtered.map(r => `
        <tr class="hover:bg-slate-50 transition">
          <td class="py-3 px-3 font-mono font-extrabold text-emerald-800">${r.id}</td>
          <td class="py-3 px-3 whitespace-nowrap">${r.tanggal}</td>
          <td class="py-3 px-3">
            <img src="${r.foto}" alt="Foto" class="w-12 h-9 object-cover rounded-lg border border-slate-200 cursor-pointer" onclick="openDetailModal('${r.id}')" />
          </td>
          <td class="py-3 px-3 font-semibold">${r.distrik} - ${r.kampung}</td>
          <td class="py-3 px-3 max-w-[190px] truncate" title="${r.namaRuas}">${r.namaRuas}</td>
          <td class="py-3 px-3">${r.jenisKerusakan}</td>
          <td class="py-3 px-3">${getSeverityBadge(r.tingkatKerusakan)}</td>
          <td class="py-3 px-3">${getStatusBadge(r.status)}</td>
          <td class="py-3 px-3 text-center whitespace-nowrap space-x-1">
            <button onclick="openDetailModal('${r.id}')" title="Detail & Verifikasi" class="px-2.5 py-1.5 rounded-lg bg-emerald-50 text-emerald-800 hover:bg-emerald-700 hover:text-white font-bold transition">
              <i class="fa-solid fa-eye"></i>
            </button>
            <button onclick="openOfficialPrintModal('${r.id}')" title="Cetak PDF Resmi" class="px-2.5 py-1.5 rounded-lg bg-amber-50 text-amber-800 hover:bg-amber-600 hover:text-white font-bold transition">
              <i class="fa-solid fa-print"></i>
            </button>
            ${currentUserRole === 'Admin' ? `
              <button onclick="deleteReportById('${r.id}')" title="Hapus Laporan (Admin)" class="px-2.5 py-1.5 rounded-lg bg-red-50 text-red-700 hover:bg-red-600 hover:text-white font-bold transition">
                <i class="fa-solid fa-trash"></i>
              </button>
            ` : ''}
          </td>
        </tr>
      `).join('');

    function saveReportStatus(reportId) {
      const newStatus = document.getElementById('modal-status-select').value;
      const idx = reports.findIndex(r => r.id === reportId);
      if (idx !== -1) {
        reports[idx].status = newStatus;
        saveReportsToStorage();
        updateAllViews();
        if (!document.getElementById('section-data').classList.contains('hidden')) {
          renderFullDataTable();
        }
        closeModal('detail-modal');
        showToast(`Status laporan ${reportId} diperbarui menjadi "${newStatus}" oleh ${currentUserRole}.`, 'success');
      }
    }

    function createColoredMarkerIcon(tingkat) {
      let color = '#059669'; // Hijau (Ringan)
      if (tingkat === 'Rusak Berat') color = '#DC2626'; // Merah
      else if (tingkat === 'Rusak Sedang') color = '#EAB308'; // Kuning

      return L.divIcon({
        className: 'custom-pin',
        html: `<div style="background:${color}; width:19px; height:19px; border-radius:9999px; border:3px solid #ffffff; box-shadow:0 3px 8px rgba(0,0,0,0.5);"></div>`,
        iconSize: [19, 19],
        iconAnchor: [9, 9]
      });
    }

    function getTileLayerByMode(mode) {
      if (mode === 'street') {
        return L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', {
          attribution: '&copy; OpenStreetMap • Kabupaten Deiyai'
        });
      }
      if (mode === 'topo') {
        return L.tileLayer('https://{s}.tile.opentopomap.org/{z}/{x}/{y}.png', {
          attribution: '&copy; OpenTopoMap • Kabupaten Deiyai'
        });
      }
      // Default: Citra Satelit Esri World Imagery
      return L.tileLayer('https://server.arcgisonline.com/ArcGIS/rest/services/World_Imagery/MapServer/tile/{z}/{y}/{x}', {
        attribution: 'Citra Satelit &copy; Esri • Kabupaten Deiyai, Papua Tengah'
      });
    }

    function initDashboardMap() {
      const mapEl = document.getElementById('dashboard-map');
      if (!mapEl || dashboardMap) return;

      dashboardMap = L.map('dashboard-map', { zoomControl: true }).setView([-4.058, 136.285], 12);
      dashboardTileLayer = getTileLayerByMode('satellite').addTo(dashboardMap);
      dashboardMarkersGroup = L.layerGroup().addTo(dashboardMap);

      // Klik pada peta satelit langsung mengisi koordinat GPS di Form Input Petugas
      dashboardMap.on('click', (e) => {
        const lat = e.latlng.lat.toFixed(6);
        const lng = e.latlng.lng.toFixed(6);
        const coordsInput = document.getElementById('adm-coords');
        if (coordsInput) coordsInput.value = `${lat}, ${lng}`;
        showToast(`Koordinat dipilih dari Peta Satelit: ${lat}, ${lng}`, 'info');
      });

      updateMapMarkers();
    }

    function changeMapLayer(mode) {
      if (!dashboardMap) return;
      if (dashboardTileLayer) dashboardMap.removeLayer(dashboardTileLayer);
      dashboardTileLayer = getTileLayerByMode(mode).addTo(dashboardMap);
    }

    function initFullMap() {
      const mapEl = document.getElementById('full-map');
      if (!mapEl) return;
      if (!fullMap) {
        fullMap = L.map('full-map').setView([-4.058, 136.285], 12);
        fullTileLayer = getTileLayerByMode('satellite').addTo(fullMap);
        fullMarkersGroup = L.layerGroup().addTo(fullMap);
      }
      fullMap.invalidateSize();
      updateMapMarkers();
    }

    function changeFullMapLayer(mode) {
      if (!fullMap) return;
      if (fullTileLayer) fullMap.removeLayer(fullTileLayer);
      fullTileLayer = getTileLayerByMode(mode).addTo(fullMap);
    }

    function updateMapMarkers() {
      const bindMarkers = (group, isDashboard) => {
        if (!group) return;
        group.clearLayers();
        reports.forEach(r => {
          const marker = L.marker([r.latitude, r.longitude], {
            icon: createColoredMarkerIcon(r.tingkatKerusakan)
          });
          const popupHtml = `
            <div style="min-width:205px; font-family:Inter,sans-serif;">
              <div style="display:flex; justify-content:space-between; align-items:center;">
                <span style="font-weight:900; color:#065f46; font-size:12px;">${r.id}</span>
                <span style="font-size:10px; font-weight:700; padding:1px 6px; border-radius:4px; background:#fef2f2; color:#dc2626;">${r.tingkatKerusakan}</span>
              </div>
              <div style="font-weight:700; font-size:11px; margin-top:3px; color:#0f172a;">${r.namaRuas}</div>
              <div style="font-size:10px; color:#475569;">Distrik ${r.distrik} - Kp. ${r.kampung}</div>
              <div style="font-size:10px; font-family:monospace; color:#64748b; margin-top:2px;">GPS: ${r.latitude}, ${r.longitude}</div>
              <img src="${r.foto}" style="width:100%; height:85px; object-fit:cover; border-radius:8px; margin-top:6px; border:1px solid #e2e8f0;" />
              <button onclick="openDetailModal('${r.id}')" style="margin-top:6px; width:100%; padding:5px; background:#065f46; color:#fff; border:none; border-radius:6px; font-size:11px; font-weight:700; cursor:pointer;">Lihat Detail Laporan</button>
            </div>
          `;
          marker.bindPopup(popupHtml);
          if (isDashboard) {
            marker.on('click', () => selectReportForDetail(r.id));
          }
          marker.addTo(group);
        });
      };

      bindMarkers(dashboardMarkersGroup, true);
      bindMarkers(fullMarkersGroup, false);
    }

    function downloadQRCode(containerId, filename) {
      const canvas = document.querySelector(`#${containerId} canvas`);
      if (!canvas) {
        showToast('QR Code belum siap diunduh.', 'warning');
        return;
      }
      const link = document.createElement('a');
      link.download = `QR-SIPJ-${filename}.png`;
      link.href = canvas.toDataURL('image/png');
      link.click();
      showToast(`QR Code ${filename} berhasil diunduh (PNG).`, 'success');
    }

    function triggerDatabaseBackup() {
      const now = new Date();
      const timeStr = `Hari ini, ${String(now.getHours()).padStart(2, '0')}:${String(now.getMinutes()).padStart(2, '0')} WIT`;
      document.getElementById('last-backup-time').textContent = timeStr;

      const backupPayload = {
        instansi: "Dinas PUPR Kabupaten Deiyai - Bidang Bina Marga",
        sistem: "SIPJ (Sistem Informasi Pelaporan Kondisi Kerusakan Jalan dan Jembatan)",
        waktu_backup: now.toISOString(),
        dibuat_oleh_peran: currentUserRole,
        total_record_aktif: reports.length,
        data_laporan: reports
      };

      const blob = new Blob([JSON.stringify(backupPayload, null, 2)], { type: 'application/json' });
      const url = URL.createObjectURL(blob);
      const a = document.createElement('a');
      a.href = url;
      a.download = `backup_sipj_deiyai_${now.toISOString().split('T')[0]}.json`;
      a.click();
      URL.revokeObjectURL(url);

      showToast('Backup database berhasil dibuat & diunduh (.JSON)!', 'success');
    }

    function exportCSV() {
      const headers = ["Nomor Tiket", "Tanggal", "Pelapor", "No HP", "Distrik", "Kampung", "Kategori", "Ruas Jalan/Jembatan", "Jenis Kerusakan", "Tingkat Kerusakan", "Latitude", "Longitude", "Status"];
      const rows = reports.map(r => [
        r.id, r.tanggal, `"${r.namaPelapor}"`, r.nomorHp, r.distrik, r.kampung, r.kategoriObjek, `"${r.namaRuas}"`, `"${r.jenisKerusakan}"`, r.tingkatKerusakan, r.latitude, r.longitude, r.status
      ]);

      const csvContent = "data:text/csv;charset=utf-8," + [headers.join(","), ...rows.map(e => e.join(","))].join("\n");
      const encodedUri = encodeURI(csvContent);
      const link = document.createElement("a");
      link.setAttribute("href", encodedUri);
      link.setAttribute("download", `REKAP-SIPJ-PUPR-DEIYAI-${new Date().toISOString().split('T')[0]}.csv`);
      document.body.appendChild(link);
      link.click();
      document.body.removeChild(link);

      showToast('Data rekapitulasi berhasil diekspor ke file CSV!', 'success');
    }

    }
